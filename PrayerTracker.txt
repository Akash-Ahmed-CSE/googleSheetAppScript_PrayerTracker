/* ---------- HELPERS ---------- */
// Deletes unused rows/columns so the sheet is exactly rows x cols
function trimSheet_(sh, rows, cols) {
  if (sh.getMaxRows() < rows) sh.insertRowsAfter(sh.getMaxRows(), rows - sh.getMaxRows());
  if (sh.getMaxColumns() < cols) sh.insertColumnsAfter(sh.getMaxColumns(), cols - sh.getMaxColumns());
  if (sh.getMaxRows() > rows) sh.deleteRows(rows + 1, sh.getMaxRows() - rows);
  if (sh.getMaxColumns() > cols) sh.deleteColumns(cols + 1, sh.getMaxColumns() - cols);
}

/* ---------- YEAR GENERATION ---------- */
function askYear_(title) {
  const ui = SpreadsheetApp.getUi();
  const res = ui.prompt(title, 'Enter a year (e.g. 2027):', ui.ButtonSet.OK_CANCEL);
  if (res.getSelectedButton() !== ui.Button.OK) return null;
  const year = parseInt(res.getResponseText(), 10);
  if (isNaN(year) || year < 2000 || year > 2100) {
    ui.alert('Please enter a valid year between 2000 and 2100.');
    return null;
  }
  return year;
}

// Creates BOTH sheets for a year: "2027" (Namaz) and "Roja 2027"
function addNewYear() {
  const year = askYear_('Add New Year');
  if (!year) return;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const rojaName = CONFIG.ROJA_PREFIX + year;
  let namaz = ss.getSheetByName(String(year));
  let roja = ss.getSheetByName(rojaName);

  if (namaz || roja) {
    const ui = SpreadsheetApp.getUi();
    const ans = ui.alert('Sheet(s) for ' + year + ' already exist',
      'Rebuild dates and formatting for "' + year + '" and "' + rojaName + '"? Your marks are kept.', ui.ButtonSet.YES_NO);
    if (ans !== ui.Button.YES) return;
  }
  if (!namaz) namaz = ss.insertSheet(String(year));
  if (!roja) roja = ss.insertSheet(rojaName);

  buildYear_(namaz, year);
  buildRojaYear_(roja, year);
  ensureDashboard_(ss, false);
  refreshTrackerDropdown_(null, true);   // dashboard jumps to the latest year
  updateDashboard();
  ss.setActiveSheet(namaz);
}

// Creates / refreshes only the Roja sheet for a year
function addRojaSheet() {
  const year = askYear_('Add / Refresh Roja Sheet');
  if (!year) return;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const name = CONFIG.ROJA_PREFIX + year;
  let sheet = ss.getSheetByName(name);
  if (!sheet) sheet = ss.insertSheet(name);
  buildRojaYear_(sheet, year);
  ensureDashboard_(ss, false);
  refreshTrackerDropdown_(null, false);
  updateDashboard();
  ss.setActiveSheet(sheet);
}

// Formats the current sheet for a year AND renames it to that year
function setupCurrentSheet() {
  const year = askYear_('Setup Current Sheet');
  if (!year) return;
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ss.getActiveSheet();
  const nm = sheet.getName();
  if (nm === CONFIG.DASH || nm === CONFIG.OLD_DATA || nm.indexOf(CONFIG.ROJA_PREFIX) === 0) {
    SpreadsheetApp.getUi().alert('Open your Namaz tracker sheet first, then run this again.');
    return;
  }
  const clash = ss.getSheetByName(String(year));
  if (clash && clash.getSheetId() !== sheet.getSheetId()) {
    SpreadsheetApp.getUi().alert('A sheet named "' + year + '" already exists. Use that one, or delete it first.');
    return;
  }
  buildYear_(sheet, year);
  sheet.setName(String(year));
  ensureDashboard_(ss, false);
  refreshTrackerDropdown_(null, true);
  updatePrayerTimes();
}

function buildYear_(sheet, year) {
  const start = new Date(year, 0, 1);
  const days = Math.round((new Date(year + 1, 0, 1) - start) / 86400000); // 365 or 366
  const lastRow = days + 1;
  const cols = CONFIG.HEADERS.length; // 17 (A:Q)

  // Make sure the grid is big enough first
  if (sheet.getMaxRows() < CONFIG.YEAR_ROWS) sheet.insertRowsAfter(sheet.getMaxRows(), CONFIG.YEAR_ROWS - sheet.getMaxRows());
  if (sheet.getMaxColumns() < cols) sheet.insertColumnsAfter(sheet.getMaxColumns(), cols - sheet.getMaxColumns());

  // Clear leftover rows below the year (e.g. leap year -> normal year)
  const maxRows = sheet.getMaxRows();
  if (maxRows > lastRow) {
    const extra = sheet.getRange(lastRow + 1, 1, maxRows - lastRow, cols);
    extra.clearContent();
    extra.clearDataValidations();
    extra.clearFormat();
  }

  // Headers
  const headerRange = sheet.getRange(1, 1, 1, cols);
  headerRange.setValues([CONFIG.HEADERS])
    .setFontWeight('bold').setFontColor('#000000')
    .setHorizontalAlignment('center').setVerticalAlignment('middle').setWrap(true);
  headerRange.setBackgrounds([CONFIG.HEADER_COLORS]);
  sheet.setRowHeight(1, 45);

  // Dates
  const dates = [];
  for (let i = 0; i < days; i++) dates.push([new Date(year, 0, 1 + i)]);
  sheet.getRange(2, 1, days, 1).setValues(dates)
    .setNumberFormat('d-mmm-yyyy').setFontWeight('normal').setHorizontalAlignment('right');

  // Dropdowns: Farz + Witr = Done/Kaza/Skipped ; Sunnah + Nafl = Done/Skipped only
  const fullRule = SpreadsheetApp.newDataValidation()
    .requireValueInList([CONFIG.DONE, CONFIG.KAZA, CONFIG.SKIPPED], true)
    .setAllowInvalid(true).build();
  const shortRule = SpreadsheetApp.newDataValidation()
    .requireValueInList([CONFIG.DONE, CONFIG.SKIPPED], true)
    .setAllowInvalid(false).build();
  const marks = sheet.getRange(2, 2, days, cols - 1);
  marks.setDataValidation(fullRule).setHorizontalAlignment('center');
  CONFIG.SUNNAH_NAFL_COLS.forEach(function (letter) {
    sheet.getRange(2, colIdx_(letter), days, 1).setDataValidation(shortRule);
  });

  // Colour rules + today highlight
  const rules = [
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo(CONFIG.DONE)
      .setBackground('#B7E1CD').setFontColor('#0B5394').setRanges([marks]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo(CONFIG.KAZA)
      .setBackground('#FCE5CD').setFontColor('#B45F06').setRanges([marks]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo(CONFIG.SKIPPED)
      .setBackground('#F4CCCC').setFontColor('#990000').setRanges([marks]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenFormulaSatisfied('=$A2=TODAY()')
      .setBackground('#FFF2CC').setBold(true).setRanges([sheet.getRange(2, 1, days, 1)]).build()
  ];
  sheet.setConditionalFormatRules(rules);

  // Layout
  sheet.setFrozenRows(1);
  sheet.setFrozenColumns(1);
  sheet.setColumnWidth(1, 110);
  sheet.setColumnWidths(2, cols - 1, 95);

  // Grid + prayer-group separators
  sheet.getRange(1, 1, lastRow, cols)
    .setBorder(true, true, true, true, true, true, '#BDBDBD', SpreadsheetApp.BorderStyle.SOLID);

  CONFIG.GROUPS.forEach(function (g, idx) {
    const c1 = g[1].charCodeAt(0) - 64;
    const c2 = g[2].charCodeAt(0) - 64;
    sheet.getRange(2, c1, days, c2 - c1 + 1).setBackground(idx % 2 === 0 ? '#FFFFFF' : '#F3F3F3');
    sheet.getRange(1, c1, lastRow, 1)
      .setBorder(null, true, null, null, null, null, '#000000', SpreadsheetApp.BorderStyle.SOLID_THICK);
  });
  sheet.getRange(1, cols, lastRow, 1)
    .setBorder(null, null, null, true, null, null, '#000000', SpreadsheetApp.BorderStyle.SOLID_THICK);

  // Delete unused rows / columns (also removes the old prayer-times panel)
  trimSheet_(sheet, CONFIG.YEAR_ROWS, cols);

  // Scroll to today if this is the current year
  const now = new Date();
  if (now.getFullYear() === year) {
    const dayOfYear = Math.floor((new Date(now.getFullYear(), now.getMonth(), now.getDate()) - start) / 86400000);
    sheet.setActiveRange(sheet.getRange(2 + dayOfYear, 2));
  }
}

/* ---------- ROJA SHEET ---------- */
function pad2_(n) { return (n < 10 ? '0' : '') + n; }

// Gets Fajr + Maghrib (minutes) for every day of a year. Returns { 'dd-mm-yyyy': {fajr, maghrib} }
function fetchYearTimings_(year) {
  const base = 'https://api.aladhan.com/v1/calendar/' + year;
  const q = '?latitude=' + CONFIG.LAT + '&longitude=' + CONFIG.LNG +
    '&method=' + CONFIG.METHOD + '&school=' + CONFIG.SCHOOL +
    '&timezonestring=' + encodeURIComponent(CONFIG.TZ);
  const map = {};

  function ingest(data) {
    let list = [];
    if (Array.isArray(data)) list = data;
    else if (data && typeof data === 'object') {
      Object.keys(data).forEach(function (k) { list = list.concat(data[k]); });
    }
    list.forEach(function (day) {
      try {
        map[day.date.gregorian.date] = {
          fajr: toMin_(day.timings.Fajr),
          maghrib: toMin_(day.timings.Maghrib)
        };
      } catch (e) { /* skip bad entry */ }
    });
  }

  // 1) whole year in one call
  try {
    const r = UrlFetchApp.fetch(base + q + '&annual=true', { muteHttpExceptions: true });
    if (r.getResponseCode() === 200) ingest(JSON.parse(r.getContentText()).data);
  } catch (e) { /* fall through */ }

  // 2) fallback: month by month
  if (Object.keys(map).length < 300) {
    for (let m = 1; m <= 12; m++) {
      try {
        const r = UrlFetchApp.fetch('https://api.aladhan.com/v1/calendar/' + year + '/' + m + q, { muteHttpExceptions: true });
        if (r.getResponseCode() === 200) ingest(JSON.parse(r.getContentText()).data);
      } catch (e) { /* skip */ }
    }
  }
  return map;
}

function buildRojaYear_(sheet, year) {
  const start = new Date(year, 0, 1);
  const days = Math.round((new Date(year + 1, 0, 1) - start) / 86400000);
  const lastRow = days + 1;
  const cols = CONFIG.ROJA_HEADERS.length; // 6 (A:F)

  if (sheet.getMaxRows() < CONFIG.YEAR_ROWS) sheet.insertRowsAfter(sheet.getMaxRows(), CONFIG.YEAR_ROWS - sheet.getMaxRows());
  if (sheet.getMaxColumns() < cols) sheet.insertColumnsAfter(sheet.getMaxColumns(), cols - sheet.getMaxColumns());

  // Leftover rows (leap year -> normal year)
  const maxRows = sheet.getMaxRows();
  if (maxRows > lastRow) {
    const extra = sheet.getRange(lastRow + 1, 1, maxRows - lastRow, cols);
    extra.clearContent(); extra.clearDataValidations(); extra.clearFormat();
  }

  // Headers
  const h = sheet.getRange(1, 1, 1, cols);
  h.setValues([CONFIG.ROJA_HEADERS]).setFontWeight('bold').setFontColor('#000000')
    .setHorizontalAlignment('center').setVerticalAlignment('middle').setWrap(true);
  h.setBackgrounds([CONFIG.ROJA_HEADER_COLORS]);
  sheet.setRowHeight(1, 40);

  // Dates + weekday
  const dow = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
  const dateRows = [], dayRows = [];
  for (let i = 0; i < days; i++) {
    const d = new Date(year, 0, 1 + i);
    dateRows.push([d]);
    dayRows.push([dow[d.getDay()]]);
  }
  sheet.getRange(2, 1, days, 1).setValues(dateRows).setNumberFormat('d-mmm-yyyy')
    .setFontWeight('normal').setHorizontalAlignment('right');
  sheet.getRange(2, 2, days, 1).setValues(dayRows).setHorizontalAlignment('center');

  // Sehri + Iftar times for the whole year
  const times = fetchYearTimings_(year);
  const tv = [];
  let found = 0;
  for (let i = 0; i < days; i++) {
    const d = new Date(year, 0, 1 + i);
    const key = pad2_(d.getDate()) + '-' + pad2_(d.getMonth() + 1) + '-' + d.getFullYear();
    const t = times[key];
    if (t) { found++; tv.push([(t.fajr + CONFIG.SEHRI_OFFSET_MIN) / 1440, (t.maghrib + CONFIG.IFTAR_OFFSET_MIN) / 1440]); }
    else tv.push(['', '']);
  }
  sheet.getRange(2, 3, days, 2).setValues(tv).setNumberFormat('h:mm AM/PM').setHorizontalAlignment('center');
  if (found < days) {
    SpreadsheetApp.getActiveSpreadsheet().toast(
      'Sehri/Iftar times found for ' + found + ' of ' + days + ' days. Run "Add / Refresh Roja Sheet" again to retry.',
      'Prayer Tracker', 8);
  }

  // Status dropdown (existing marks are kept)
  const status = sheet.getRange(2, 5, days, 1);
  status.setDataValidation(SpreadsheetApp.newDataValidation()
    .requireValueInList([CONFIG.DONE, CONFIG.SKIPPED, CONFIG.LATER], true)
    .setAllowInvalid(false).build()).setHorizontalAlignment('center');

  // Colour rules + today highlight
  sheet.setConditionalFormatRules([
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo(CONFIG.DONE)
      .setBackground('#B7E1CD').setFontColor('#0B5394').setRanges([status]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo(CONFIG.LATER)
      .setBackground('#CFE2F3').setFontColor('#1155CC').setRanges([status]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenTextEqualTo(CONFIG.SKIPPED)
      .setBackground('#F4CCCC').setFontColor('#990000').setRanges([status]).build(),
    SpreadsheetApp.newConditionalFormatRule().whenFormulaSatisfied('=$A2=TODAY()')
      .setBackground('#FFF2CC').setBold(true).setRanges([sheet.getRange(2, 1, days, 4)]).build()
  ]);

  // Layout
  sheet.setFrozenRows(1);
  sheet.setFrozenColumns(1);
  sheet.setColumnWidth(1, 110); sheet.setColumnWidth(2, 60);
  sheet.setColumnWidth(3, 100); sheet.setColumnWidth(4, 100);
  sheet.setColumnWidth(5, 130); sheet.setColumnWidth(6, 220);
  sheet.getRange(1, 1, lastRow, cols)
    .setBorder(true, true, true, true, true, true, '#BDBDBD', SpreadsheetApp.BorderStyle.SOLID);
  sheet.setTabColor('#B4A7D6');

  // Delete unused rows / columns
  trimSheet_(sheet, CONFIG.YEAR_ROWS, cols);

  const now = new Date();
  if (now.getFullYear() === year) {
    const dayOfYear = Math.floor((new Date(now.getFullYear(), now.getMonth(), now.getDate()) - start) / 86400000);
    sheet.setActiveRange(sheet.getRange(2 + dayOfYear, 5));
  }
}

/* ---------- KAZA RULE ----------
   Farz = Kaza  ->  linked Sunnah/Nafl cells become Skipped,
   but ONLY if those cells are currently empty. */
function colIdx_(letter) {
  return letter.charCodeAt(0) - 64;
}

// Simple trigger: runs automatically on every edit (no install needed)
function onEdit(e) {
  if (!e || !e.range) return;
  const sh = e.range.getSheet();
  if (!/^\d{4}$/.test(sh.getName())) return;       // Namaz year sheets only
  const r = e.range;
  if (r.getRow() < 2) return;
  if (r.getNumRows() * r.getNumColumns() > 200) return; // ignore huge pastes; use the menu item instead

  const vals = r.getValues();
  for (let i = 0; i < vals.length; i++) {
    for (let j = 0; j < vals[i].length; j++) {
      if (vals[i][j] !== CONFIG.KAZA) continue;
      const letter = String.fromCharCode(64 + r.getColumn() + j);
      const deps = CONFIG.KAZA_LINKS[letter];
      if (!deps) continue;
      const row = r.getRow() + i;
      deps.forEach(function (d) {
        const cell = sh.getRange(d + row);
        if (cell.getValue() === '') cell.setValue(CONFIG.SKIPPED);   // only fill empty cells
      });
    }
  }
}

// On demand: apply the rule to every row of the current sheet (empty cells only)
function applyKazaRule() {
  const sh = SpreadsheetApp.getActiveSheet();
  if (!/^\d{4}$/.test(sh.getName())) {
    SpreadsheetApp.getUi().alert('Open a Namaz year sheet (e.g. 2026) first.');
    return;
  }
  const last = sh.getLastRow();
  if (last < 2) return;
  const rng = sh.getRange(2, 1, last - 1, CONFIG.HEADERS.length);
  const v = rng.getValues();
  let changed = 0;
  v.forEach(function (row) {
    Object.keys(CONFIG.KAZA_LINKS).forEach(function (farz) {
      if (row[colIdx_(farz) - 1] !== CONFIG.KAZA) return;
      CONFIG.KAZA_LINKS[farz].forEach(function (d) {
        const k = colIdx_(d) - 1;
        if (row[k] === '') { row[k] = CONFIG.SKIPPED; changed++; }
      });
    });
  });
  rng.setValues(v);
  SpreadsheetApp.getActiveSpreadsheet().toast(changed + ' empty cell(s) set to Skipped.', 'Prayer Tracker', 5);
}

/* ---------- FORMULA HELPERS ---------- */
// Reference a range on the Namaz year sheet chosen in Dashboard!C3
function rng_(a1) {
  return `INDIRECT("'"&$C$3&"'!${a1}")`;
}
// Reference a range on the Roja sheet of the chosen year
function rj_(a1) {
  return `INDIRECT("'${CONFIG.ROJA_PREFIX}"&$C$3&"'!${a1}")`;
}

// Farz Done / Farz Marked formula parts, limited by a mask (use '1' for the whole year)
function farzParts_(mask) {
  const done = CONFIG.FARZ_COLS.map(function (c) {
    return 'SUMPRODUCT(' + mask + '*(' + rng_(c + '2:' + c + '367') + '="' + CONFIG.DONE + '"))';
  }).join('+');
  const marked = CONFIG.FARZ_COLS.map(function (c) {
    return 'SUMPRODUCT(' + mask + '*(' + rng_(c + '2:' + c + '367') + '<>""))';
  }).join('+');
  return { done: done, marked: marked };
}
function farzPct_(mask) {
  const p = farzParts_(mask);
  return '=IFERROR(IF((' + p.marked + ')=0,NA(),(' + p.done + ')/(' + p.marked + ')),NA())';
}

// Array of 1/0 per day: all Farz Done
function perfectArr_() {
  return CONFIG.FARZ_COLS.map(function (c) {
    return '(' + rng_(c + '2:' + c + '367') + '="' + CONFIG.DONE + '")';
  }).join('*');
}
// Array of TRUE/FALSE per day: at least one Farz marked
function markedArr_() {
  return '(' + CONFIG.FARZ_COLS.map(function (c) {
    return '(' + rng_(c + '2:' + c + '367') + '<>"")';
  }).join('+') + ')>0';
}
const SCAN_STREAK_ = 'SCAN(0,p,LAMBDA(a,v,IF(v=1,a+1,0)))';

/* ---------- CHART DATA BLOCK (on the Dashboard, columns P:Y) ---------- */
// Protects the chart data block (P4:Y59) so it is read-only
function lockChartData_(sh) {
  const DESC = 'Chart data (read only)';

  // Remove old locks first so they don't pile up every time the dashboard is rebuilt
  sh.getProtections(SpreadsheetApp.ProtectionType.RANGE).forEach(function (p) {
    if (p.getDescription() === DESC) p.remove();
  });

  const p = sh.getRange('P4:Y59').protect().setDescription(DESC);

  if (CONFIG.CHART_LOCK === 'strict') {
    const me = Session.getEffectiveUser();
    p.addEditor(me);
    p.removeEditors(p.getEditors().filter(function (u) {
      return u.getEmail() !== me.getEmail();
    }));
    if (p.canDomainEdit()) p.setDomainEdit(false);
  } else {
    p.setWarningOnly(true);
  }
}

function buildChartData_(sh) {
  const months = ['Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun', 'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec'];
  const hdr = function (a1, vals) {
    sh.getRange(a1).setValues([vals]).setFontWeight('bold').setBackground('#EFEFEF');
  };

  sh.getRange('P4:Y4').merge().setValue('Chart data (used by the graphs — do not edit)')
    .setFontStyle('italic').setHorizontalAlignment('left');

  // Weekly Farz % (P:Q) rows 7..59
  hdr('P6:Q6', ['Week', 'Farz Done %']);
  const A = rng_('A2:A367');
  const wkNum = [], wkF = [];
  for (let w = 1; w <= 53; w++) {
    wkNum.push([w]);
    wkF.push([farzPct_('(INT((' + A + '-DATE(YEAR(' + A + '),1,1))/7)+1=' + w + ')')]);
  }
  sh.getRange(7, 16, 53, 1).setValues(wkNum);
  sh.getRange(7, 17, 53, 1).setFormulas(wkF).setNumberFormat('0%');

  // Monthly Farz % (S:T) rows 7..18

// Year-named Namaz sheets only (e.g. "2026"), newest first
function yearSheetNames_(ss) {
  return ss.getSheets().map(function (s) { return s.getName(); })
    .filter(function (n) { return /^\d{4}$/.test(n); })
    .sort(function (a, b) { return b - a; });
}

// Fills the Report Year dropdown. toLatest = true jumps to the newest year.
function refreshTrackerDropdown_(dash, toLatest) {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  dash = dash || ss.getSheetByName(CONFIG.DASH);
  if (!dash) return;
  const names = yearSheetNames_(ss);
  const cell = dash.getRange('C3');
  if (!names.length) {
    cell.clearDataValidations();
    cell.clearContent();
    return;
  }
  cell.setDataValidation(SpreadsheetApp.newDataValidation()
    .requireValueInList(names, true).setAllowInvalid(false).build());
  const cur = String(cell.getValue());
  if (toLatest || names.indexOf(cur) === -1) cell.setValue(Number(names[0]));
}

/* ---------- UPDATE (one button: redraws everything, keeps your year) ---------- */
function updateDashboard() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const old = ss.getSheetByName(CONFIG.DASH);
  const keep = old ? String(old.getRange('C3').getValue()) : null;

  const dash = ensureDashboard_(ss, true);
  if (keep && ss.getSheetByName(keep)) dash.getRange('C3').setValue(Number(keep));
  SpreadsheetApp.flush();

  updatePrayerTimes();
  ss.setActiveSheet(dash);
  ss.toast('Dashboard updated ✅', 'Prayer Tracker', 4);
}

// Installable on-edit trigger: ticking the checkbox in F3 runs the update
function onDashboardEdit(e) {
  if (!e || !e.range) return;
  const sh = e.range.getSheet();
  if (sh.getName() !== CONFIG.DASH) return;
  if (e.range.getA1Notation() === 'F3' && e.range.getValue() === true) {
    try {
      updateDashboard();
    } finally {
      const d = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(CONFIG.DASH);
      if (d) d.getRange('F3').setValue(false);   // reset the "button"
    }
  }
}

/* ---------- TODAY'S PRAYER TIMES + SEHRI/IFTAR (written to the Dashboard) ---------- */
function updatePrayerTimes() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const sheet = ensureDashboard_(ss, false);
  const dateStr = Utilities.formatDate(new Date(), CONFIG.TZ, 'dd-MM-yyyy');

  const url = 'https://api.aladhan.com/v1/timings/' + dateStr +
    '?latitude=' + CONFIG.LAT + '&longitude=' + CONFIG.LNG +
    '&method=' + CONFIG.METHOD + '&school=' + CONFIG.SCHOOL +
    '&timezonestring=' + encodeURIComponent(CONFIG.TZ);

  let t;
  try {
    const resp = UrlFetchApp.fetch(url, { muteHttpExceptions: true });
    if (resp.getResponseCode() !== 200) throw new Error('HTTP ' + resp.getResponseCode());
    t = JSON.parse(resp.getContentText()).data.timings;
  } catch (e) {
    ss.toast('Could not fetch prayer times: ' + e.message, 'Prayer Tracker', 8);
    return;
  }

  const m = {
    fajr: toMin_(t.Fajr), sunrise: toMin_(t.Sunrise), zuhr: toMin_(t.Dhuhr),
    asr: toMin_(t.Asr), maghrib: toMin_(t.Maghrib), isha: toMin_(t.Isha)
  };

  sheet.getRange('B5').setValue('🕌 Prayer Times - ' + dateStr);
  sheet.getRange('C6:C11').setValues([
    [m.fajr / 1440], [m.sunrise / 1440], [m.zuhr / 1440],
    [m.asr / 1440], [m.maghrib / 1440], [m.isha / 1440]
  ]).setNumberFormat('h:mm AM/PM');

  sheet.getRange('C15:C17').setValues([
    [fmt12_(m.sunrise) + ' - ' + fmt12_(m.sunrise + CONFIG.SUNRISE_BLOCK)],
    [fmt12_(m.zuhr - CONFIG.NOON_BLOCK) + ' - ' + fmt12_(m.zuhr)],
    [fmt12_(m.maghrib - CONFIG.MAGHRIB_BLOCK) + ' - ' + fmt12_(m.maghrib)]
  ]);

  // Roja: today's Sehri end and Iftar
  sheet.getRange('C20:C21').setValues([
    [(m.fajr + CONFIG.SEHRI_OFFSET_MIN) / 1440],
    [(m.maghrib + CONFIG.IFTAR_OFFSET_MIN) / 1440]
  ]).setNumberFormat('h:mm AM/PM');
}

function toMin_(s) {
  const p = String(s).match(/(\d{1,2}):(\d{2})/);
  return parseInt(p[1], 10) * 60 + parseInt(p[2], 10);
}

function fmt12_(mins) {
  mins = ((mins % 1440) + 1440) % 1440;
  let h = Math.floor(mins / 60);
  const mm = mins % 60;
  const ap = h >= 12 ? 'PM' : 'AM';
  h = h % 12; if (h === 0) h = 12;
  return h + ':' + (mm < 10 ? '0' : '') + mm + ' ' + ap;
}

/* ---------- TRIGGERS (daily refresh, on-open refresh, button) ---------- */
function installTriggers() {
  const ss = SpreadsheetApp.getActive();
  ScriptApp.getProjectTriggers().forEach(function (tr) {
    const fn = tr.getHandlerFunction();
    if (fn === 'updatePrayerTimes' || fn === 'refreshOnOpen' || fn === 'onDashboardEdit') ScriptApp.deleteTrigger(tr);
  });
  ScriptApp.newTrigger('updatePrayerTimes').timeBased().everyDays(1).atHour(0).nearMinute(5)
    .inTimezone(CONFIG.TZ).create();
  ScriptApp.newTrigger('refreshOnOpen').forSpreadsheet(ss).onOpen().create();
  ScriptApp.newTrigger('onDashboardEdit').forSpreadsheet(ss).onEdit().create();
  ss.toast('Auto-update and Update button installed.', 'Prayer Tracker', 5);
  updateDashboard();
}

// Installable on-open trigger (simple onOpen can't call web APIs)
function refreshOnOpen() {
  updatePrayerTimes();
}
